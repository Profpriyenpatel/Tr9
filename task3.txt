server.js

const express = require("express");
const mongoose = require("mongoose");
const cors = require("cors");

const app = express();

app.use(cors());
app.use(express.json());

// MongoDB Connection
mongoose.connect("mongodb://127.0.0.1:27017/productdb")
.then(() => console.log("MongoDB Connected"))
.catch((err) => console.log(err));

// Schema
const productSchema = new mongoose.Schema({
    productId: {
        type: String,
        required: true
    },
    date: {
        type: Date,
        required: true
    },
    message: {
        type: String,
        required: true,
        validate: {
            validator: function (value) {
                return value.trim().split(/\s+/).length <= 150;
            },
            message: "Message cannot exceed 150 words."
        }
    }
});

// Model
const Product = mongoose.model("Product", productSchema);

// Insert Data
app.post("/product", async (req, res) => {

    try {

        const product = await Product.create(req.body);

        res.status(201).json({
            success: true,
            message: "Data Inserted Successfully",
            data: product
        });

    } catch (err) {

        res.status(400).json({
            success: false,
            message: err.message
        });

    }

});

// Start Server
app.listen(5000, () => {
    console.log("Server running on port 5000");
});

import { useState } from "react";
import axios from "axios";

function Product() {

    const [productId, setProductId] = useState("");
    const [date, setDate] = useState("");
    const [message, setMessage] = useState("");

    const handleSubmit = async (e) => {

        e.preventDefault();

        const words = message.trim() === "" ? [] : message.trim().split(/\s+/);

        if (words.length > 150) {
            alert("Message cannot exceed 150 words.");
            return;
        }

        try {

            await axios.post("http://localhost:5000/product", {
                productId,
                date,
                message
            });

            alert("Data Inserted Successfully");

            setProductId("");
            setDate("");
            setMessage("");

        } catch (err) {

            alert(err.response?.data?.message || "Error");

        }

    };

    return (

        <div style={{ width: "450px", margin: "40px auto" }}>

            <h2>Product Form</h2>

            <form onSubmit={handleSubmit}>

                <label>Product ID</label><br />

                <select
                    value={productId}
                    onChange={(e) => setProductId(e.target.value)}
                    required
                >
                    <option value="">Select Product</option>
                    <option value="P101">P101</option>
                    <option value="P102">P102</option>
                    <option value="P103">P103</option>
                    <option value="P104">P104</option>
                </select>

                <br /><br />

                <label>Date</label><br />

                <input
                    type="date"
                    value={date}
                    onChange={(e) => setDate(e.target.value)}
                    required
                />

                <br /><br />

                <label>Message</label><br />

                <textarea
                    rows="6"
                    cols="40"
                    value={message}
                    onChange={(e) => setMessage(e.target.value)}
                    required
                />

                <br />

                <p>
                    Word Count:{" "}
                    {message.trim() === ""
                        ? 0
                        : message.trim().split(/\s+/).length}
                    /150
                </p>

                <button type="submit">Submit</button>

            </form>

        </div>

    );
}

export default Product;
