import React, { useState, useEffect } from "react";
import axios from "axios";
import "bootstrap/dist/css/bootstrap.min.css";

function WeatherCard() {
  const [weather, setWeather] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    axios
      .get(
        "https://api.open-meteo.com/v1/forecast?latitude=23.0225&longitude=72.5714&current=temperature_2m,relative_humidity_2m,wind_speed_10m"
      )
      .then((response) => {
        setWeather(response.data.current);
        setLoading(false);
      })
      .catch((err) => {
        setError("Unable to fetch weather data");
        setLoading(false);
      });
  }, []);

  return (
    <div className="container mt-5">
      <div className="row justify-content-center">
        <div className="col-md-5">

          <div className="card shadow-lg border-0 rounded-4">
            <div className="card-header bg-primary text-white text-center py-3">
              <h3>🌤 Ahmedabad Weather</h3>
            </div>

            <div className="card-body p-4">

              {loading && (
                <div className="text-center">
                  <div
                    className="spinner-border text-primary"
                    role="status"
                  ></div>
                  <p className="mt-3">Loading weather data...</p>
                </div>
              )}

              {error && (
                <div className="alert alert-danger text-center">
                  {error}
                </div>
              )}

              {!loading && !error && weather && (
                <>
                  <div className="text-center mb-4">
                    <h1>{weather.temperature_2m}°C</h1>
                    <p className="text-muted">
                      Current Temperature
                    </p>
                  </div>

                  <div className="row text-center">

                    <div className="col-6 mb-3">
                      <div className="card bg-light">
                        <div className="card-body">
                          <h5>💧 Humidity</h5>
                          <h4>
                            {weather.relative_humidity_2m}%
                          </h4>
                        </div>
                      </div>
                    </div>

                    <div className="col-6 mb-3">
                      <div className="card bg-light">
                        <div className="card-body">
                          <h5>🌬 Wind</h5>
                          <h4>
                            {weather.wind_speed_10m} km/h
                          </h4>
                        </div>
                      </div>
                    </div>

                  </div>

                  <div className="alert alert-info text-center mt-3">
                    Data fetched successfully using
                    <strong> Axios + useEffect</strong>
                  </div>
                </>
              )}

            </div>
          </div>

        </div>
      </div>
    </div>
  );
}

export default WeatherCard;
